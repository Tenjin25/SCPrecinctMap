    // Mobile viewport height fallback for browsers that don't support dvh/svh well (older Android Chrome/WebView).
    // Keeps fixed panels from "jumping" when the URL bar collapses/expands.
    (function () {
      function setAppHeight() {
        try { document.documentElement.style.setProperty('--app-height', `${window.innerHeight}px`); } catch (_) {}
      }

      function setAndroidMobileFlags() {
        try {
          const root = document.documentElement;
          const ua = navigator.userAgent || '';
          const isAndroid = /Android/i.test(ua);
          const isVivaldi = /\bVivaldi\b/i.test(ua);
          const isMobile = !!(window.matchMedia && window.matchMedia('(max-width: 768px)').matches);
          root.classList.toggle('android-device', isAndroid);
          root.classList.toggle('android-mobile', isAndroid && isMobile);
          root.classList.toggle('browser-vivaldi', isVivaldi);
        } catch (_) {}
      }

      function setAndroidViewportInsets() {
        try {
          const root = document.documentElement;
          if (!root.classList.contains('android-mobile')) {
            root.style.removeProperty('--android-vv-top');
            root.style.removeProperty('--android-vv-bottom');
            return;
          }
          const vv = window.visualViewport;
          const topInset = vv ? Math.max(0, Math.round(vv.offsetTop || 0)) : 0;
          const bottomInset = vv ? Math.max(0, Math.round(window.innerHeight - ((vv.height || 0) + (vv.offsetTop || 0)))) : 0;
          root.style.setProperty('--android-vv-top', `${topInset}px`);
          root.style.setProperty('--android-vv-bottom', `${bottomInset}px`);
        } catch (_) {}
      }

      function refreshMobileViewportState() {
        setAppHeight();
        setAndroidMobileFlags();
        setAndroidViewportInsets();
        try {
          if (typeof window.updateMobileOverlayOffsets === 'function') window.updateMobileOverlayOffsets();
        } catch (_) {}
      }

      refreshMobileViewportState();
      window.addEventListener('resize', refreshMobileViewportState, { passive: true });
      window.addEventListener('orientationchange', refreshMobileViewportState, { passive: true });
      if (window.visualViewport) {
        window.visualViewport.addEventListener('resize', refreshMobileViewportState, { passive: true });
        window.visualViewport.addEventListener('scroll', setAndroidViewportInsets, { passive: true });
      }
    })();

    document.addEventListener('DOMContentLoaded', function() {
      function isMobileViewport() {
        try {
          return !!(window.matchMedia && window.matchMedia('(max-width: 768px)').matches);
        } catch (_) {
          return window.innerWidth <= 768;
        }
      }

      function syncMobileTopbarToggle() {
        try {
          const topbar = document.getElementById('map-topbar');
          const toggle = document.getElementById('mobile-topbar-toggle');
          if (!topbar || !toggle) return;
          if (!isMobileViewport()) {
            topbar.classList.remove('topbar-expanded');
          }
          const expanded = isMobileViewport() && topbar.classList.contains('topbar-expanded');
          toggle.textContent = expanded ? 'Hide details' : 'Show details';
          toggle.setAttribute('aria-expanded', expanded ? 'true' : 'false');
        } catch (_) {}
      }

      function collapseMobileTopbar() {
        try {
          if (!isMobileViewport()) return;
          const topbar = document.getElementById('map-topbar');
          if (!topbar) return;
          topbar.classList.remove('topbar-expanded');
          syncMobileTopbarToggle();
          if (typeof window.syncInitialTopbarOffset === 'function') {
            window.syncInitialTopbarOffset();
          }
        } catch (_) {}
      }

          const topbar = document.getElementById('map-topbar');
          if (!topbar) return;
          const minTopbarHeight = isMobileViewport() ? 0 : 224;

          if (typeof window.updateMobileOverlayOffsets === 'function') {
            window.updateMobileOverlayOffsets();
          }
        } catch (_) {}
      }
