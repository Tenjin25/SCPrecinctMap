          if (!dragState) return;
          const touches = event.changedTouches;
          if (!touches || !touches.length) return;
          let matched = false;
          for (let i = 0; i < touches.length; i += 1) {
            const touch = touches[i];
            if (dragState.pointerId !== null && touch.identifier !== dragState.pointerId) continue;
            updateDrag(touch.clientX, touch.clientY, event.timeStamp);
            matched = true;
            break;
          }
          if (!matched) resetDragState();
        }, { passive: true });
        grab.addEventListener('touchend', (event) => {
          if (!dragState) return;
          const touches = event.changedTouches;
          if (!touches || !touches.length) {
            resetDragState();
            return;
          }
          let matched = false;
          for (let i = 0; i < touches.length; i += 1) {
            const touch = touches[i];
            if (dragState.pointerId !== null && touch.identifier !== dragState.pointerId) continue;
            finishDrag(touch.clientX, touch.clientY, event.timeStamp);
            matched = true;
            break;
          }
          if (!matched) resetDragState();
        }, { passive: true });
        grab.addEventListener('touchcancel', () => {
          resetDragState();
        }, { passive: true });
      }

      grab.addEventListener('click', (event) => {
        if (grab.dataset.swiped === 'true') {
          event.preventDefault();
          event.stopPropagation();
          return;
        }
        event.preventDefault();
        event.stopPropagation();
        toggleMobileSheet(key);
      });
      panel.insertBefore(grab, panel.firstChild);
    }

      const panel = getMobileSheetPanel(key);

    function openMobileSheet(key, snap = 'half') {
      if (!isMobileViewportNow()) return;
      const panel = getMobileSheetPanel(key);
      if (!panel) return;
      try {
        if (mobileSheetVoteCounterSnapshot === null) {
          const counter = document.getElementById('vote-counter');
          mobileSheetVoteCounterSnapshot = !!(counter && !counter.classList.contains('minimized'));
        }
        setVoteCounterMinimizedState(true);
      } catch (_) {}
      try { closeHoverUI({ force: true }); } catch (_) {}
      Object.keys(MOBILE_SHEET_PANEL_IDS).forEach(otherKey => {
        if (otherKey === key) return;
        closeMobileSheet(otherKey);
      });
      panel.classList.add('mobile-sheet-panel', 'mobile-sheet-active');
      setMobileSheetSnap(panel, snap);
      if (key === 'search') {
        try {
          const topbar = document.getElementById('map-topbar');
          if (topbar) topbar.classList.add('topbar-expanded');
          if (typeof window.syncMobileTopbarToggle === 'function') window.syncMobileTopbarToggle();
        } catch (_) {}
      } else if (key === 'layers') {
        expandMainControlsOnMobile();
      } else if (key === 'legend') {
        expandLegendOnMobile();
      }
      if (snap === 'collapsed') {
        closeMobileSheet(key);
        activeMobileSheetKey = null;
      } else {
        activeMobileSheetKey = key;
      }
      syncMobileDockButtons();
      try { updateMobileOverlayOffsets(); } catch (_) {}
    }

    function closeAllMobileSheets() {
      Object.keys(MOBILE_SHEET_PANEL_IDS).forEach(key => closeMobileSheet(key));

    function swipeMobileSheet(key, direction = 'up') {
      if (!isMobileViewportNow()) return;
      if (!MOBILE_SHEET_PANEL_IDS[key]) return;
      if (activeMobileSheetKey !== key) {
        if (direction === 'up') openMobileSheet(key, 'half');
        return;
      }
      const current = getMobileSheetSnap(key);
      let next = current;
      if (direction === 'up') {
        next = current === 'collapsed' ? 'half' : 'full';
      } else {
        next = current === 'full' ? 'half' : 'collapsed';
      }
      if (next === 'collapsed') {
        closeMobileSheet(key);
        activeMobileSheetKey = null;
        syncMobileDockButtons();
        try { updateMobileOverlayOffsets(); } catch (_) {}
        return;
      }
      openMobileSheet(key, next);
    }

    function toggleMobileSheet(key) {
      if (!isMobileViewportNow()) return;
      if (!MOBILE_SHEET_PANEL_IDS[key]) return;
      if (activeMobileSheetKey !== key) {
        openMobileSheet(key, 'half');
        return;
      }
      const current = getMobileSheetSnap(key);
      const idx = MOBILE_SHEET_SNAP_SEQUENCE.indexOf(current);
      const next = MOBILE_SHEET_SNAP_SEQUENCE[(idx + 1) % MOBILE_SHEET_SNAP_SEQUENCE.length];
      if (next === 'collapsed') {
        closeMobileSheet(key);
        activeMobileSheetKey = null;
        syncMobileDockButtons();
        try { updateMobileOverlayOffsets(); } catch (_) {}
        return;
      }
      openMobileSheet(key, next);
    }

      if (mobileSheetSystemReady) return;

            toggleMobileSheet(key);
          });
        });
        const scrim = document.getElementById('mobile-sheet-scrim');
        if (scrim && scrim.dataset.mobileSheetReady !== 'true') {
          scrim.dataset.mobileSheetReady = 'true';
          scrim.addEventListener('click', () => { closeAllMobileSheets(); });
        }
        if (!window.__mobileSheetEscapeReady) {
          window.__mobileSheetEscapeReady = true;
          document.addEventListener('keydown', (event) => {
            if (event.key !== 'Escape') return;
            if (!isMobileViewportNow()) return;
            if (!activeMobileSheetKey) return;
            event.preventDefault();
            closeAllMobileSheets();
          });
        }
        window.addEventListener('resize', () => {
          try {
            if (!isMobileViewportNow()) {
              closeAllMobileSheets();
              document.body.classList.remove('mobile-sheet-open', 'mobile-input-active');
              mobileInputFocusSnapshot = null;
              if (mobileInputRestoreTimer) {
                clearTimeout(mobileInputRestoreTimer);
                mobileInputRestoreTimer = null;
              }
              Object.keys(MOBILE_SHEET_PANEL_IDS).forEach(key => {
                const panel = getMobileSheetPanel(key);
                if (!panel) return;
                panel.classList.remove('mobile-sheet-active');
                setMobileSheetSnap(panel, 'collapsed');
              });

              Object.keys(MOBILE_SHEET_PANEL_IDS).forEach(key => {

                panel.classList.add('mobile-sheet-panel');
                ensureMobileSheetGrab(panel, key);
                if (!panel.dataset.mobileSnap) setMobileSheetSnap(panel, 'collapsed');
              });
            }
          } catch (_) {}
        }, { passive: true });
      } catch (_) {}

    }

    function isMobileEditableField(node) {
      if (!node || !node.tagName) return false;
      const tag = node.tagName.toLowerCase();
      if (tag !== 'input' && tag !== 'textarea' && tag !== 'select') return false;
      return !!(node.closest && (node.closest('#map-topbar') || node.closest('#main-controls')));
    }

    function setupMobileInputFocusBehavior() {
      if (mobileInputFocusWired) return;
      mobileInputFocusWired = true;
      document.addEventListener('focusin', (event) => {
        const target = event.target;
        if (!isMobileViewportNow()) return;
        if (!isMobileEditableField(target)) return;
        if (!mobileInputFocusSnapshot) {
          const counter = document.getElementById('vote-counter');
          mobileInputFocusSnapshot = {
            voteCounterWasMinimized: !!(counter && counter.classList.contains('minimized')),
            activeSheetKey: activeMobileSheetKey,
            activeSheetSnap: activeMobileSheetKey ? getMobileSheetSnap(activeMobileSheetKey) : 'collapsed'
          };
        }
        document.body.classList.add('mobile-input-active');
        try { minimizeLegendOnMobile(); } catch (_) {}
        try { setVoteCounterMinimizedState(true); } catch (_) {}
        if (target.closest && target.closest('#map-topbar')) {
          openMobileSheet('search', 'half');
        }
      }, true);
      document.addEventListener('focusout', () => {
        if (!isMobileViewportNow()) return;
        if (mobileInputRestoreTimer) clearTimeout(mobileInputRestoreTimer);
        mobileInputRestoreTimer = setTimeout(() => {
          const active = document.activeElement;
          if (isMobileEditableField(active)) return;
          restoreMobileInputFocusMode();
          mobileInputRestoreTimer = null;
        }, 140);
      }, true);
    }

    function updateMobileOverlayOffsets() {
      // Measure the floating UI so mobile overlays and the desktop right rail can avoid collisions.
      try {
        const isMobile = window.matchMedia && window.matchMedia('(max-width: 768px)').matches;
        const isDesktop = window.matchMedia && window.matchMedia('(min-width: 769px)').matches;
        const root = document.documentElement;
        const topbar = document.getElementById('map-topbar');
        const voteCounter = document.getElementById('vote-counter');
        const mobileDock = document.getElementById('mobile-dock');
        const topbarVisible = !!(topbar && (!isMobile || topbar.classList.contains('mobile-sheet-active')));
        const th = topbarVisible ? Math.round(topbar.getBoundingClientRect().height || 0) : 0;
        const dockVisible = !!(isMobile && mobileDock && mobileDock.getClientRects().length);
        const dh = dockVisible ? Math.round(mobileDock.getBoundingClientRect().height || 0) : 0;
        root.style.setProperty('--topbar-h', `${Math.max(0, th)}px`);
        root.style.setProperty('--mobile-dock-h', `${Math.max(0, dh)}px`);
        if (isDesktop) {
          const counterVisible = !!(voteCounter && voteCounter.style.display !== 'none' && voteCounter.getClientRects().length);
          const vh = counterVisible ? Math.round(voteCounter.getBoundingClientRect().height) : 0;
          root.style.setProperty('--desktop-vote-counter-h', `${Math.max(0, vh)}px`);
        } else {
          root.style.removeProperty('--desktop-vote-counter-h');
        }
        if (!isMobile) {
          root.style.removeProperty('--legend-h');
          root.style.removeProperty('--controls-h');
          root.style.removeProperty('--mobile-dock-h');
          root.style.removeProperty('--vote-counter-h');
          return;
        }
        const counterVisibleMobile = !!(voteCounter && voteCounter.style.display !== 'none' && voteCounter.getClientRects().length);
        const vhm = counterVisibleMobile ? Math.round(voteCounter.getBoundingClientRect().height || 0) : 0;
        root.style.setProperty('--vote-counter-h', `${Math.max(0, vhm)}px`);
        const legend = document.getElementById('legend');
        const controls = document.getElementById('main-controls');
        const legendVisible = !!(legend && legend.classList.contains('mobile-sheet-active'));
        const controlsVisible = !!(controls && controls.classList.contains('mobile-sheet-active'));
        const lh = legendVisible ? Math.round(legend.getBoundingClientRect().height) : 56;
        const ch = controlsVisible ? Math.round(controls.getBoundingClientRect().height) : 0;
        root.style.setProperty('--legend-h', `${Math.max(0, lh)}px`);
        root.style.setProperty('--controls-h', `${Math.max(0, ch)}px`);
      } catch (_) {}
    }

    let mobileOverlayOffsetsWired = false;
    function wireMobileOverlayOffsets() {
      if (mobileOverlayOffsetsWired) return;
      mobileOverlayOffsetsWired = true;
      try {
        updateMobileOverlayOffsets();
        window.addEventListener('resize', () => { try { updateMobileOverlayOffsets(); } catch (_) {} }, { passive: true });
        window.addEventListener('orientationchange', () => { try { updateMobileOverlayOffsets(); } catch (_) {} }, { passive: true });
        if (typeof ResizeObserver !== 'undefined') {
          const targets = ['map-topbar', 'main-controls', 'legend', 'vote-counter', 'mobile-dock']
            .map(id => document.getElementById(id))
            .filter(Boolean);
          if (targets.length) {
            if (window.__overlayOffsetsResizeObserver) {
              try { window.__overlayOffsetsResizeObserver.disconnect(); } catch (_) {}
            }
            window.__overlayOffsetsResizeObserver = new ResizeObserver(() => {
              try { updateMobileOverlayOffsets(); } catch (_) {}
            });
            targets.forEach(node => {
              try { window.__overlayOffsetsResizeObserver.observe(node); } catch (_) {}
            });
          }
        }
      } catch (_) {}
    }
